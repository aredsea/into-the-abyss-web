// 전투 한 판을 정책 봇으로 끝까지(아이템은 쓰지 않는다)
async function playOut(max = 900) {
  for (let i = 0; i < max; i++) {
    const s = bv()
    if (s.over) return s
    const c = await L(`local st = B.load("key"); local c = ZZ.choose(st, "smart"); local t = c[2]; if t and st.u[t].al ~= 1 then t = nil end; return { c[1], t }`)
    await press('sk_' + c[0])
    if (bv().pend > 0) {
      const tg = c[1] || await L(`local st = B.load("key"); return ZZ.foes(st)[1]`)
      await press('tg_' + tg)
    }
  }
  return bv()
}
// 이 씨앗이면 탐색 1~5번이 어떤 결과인지
const kindsOf = async (seed, floor = 1) => await L(`local o, el = {}, 0 for s = 1, 5 do local r = B.ex_roll(${seed}, ${floor}, s, el); o[#o + 1] = r.kind; if r.kind == "elite" then el = el + 1 end end return o`)
// 원하는 결과 순서(앞 몇 칸)가 나오는 씨앗 찾기
async function seedFor(want, floor = 1) {
  for (let s = 1; s < 4000; s++) {
    const k = await kindsOf(s, floor)
    if (want.every((w, i) => w === '*' || (w === 'loot' ? ['coin', 'item', 'equip'].includes(k[i]) : (w === 'battle' ? ['battle', 'elite'].includes(k[i]) : k[i] === w)))) return s
  }
  throw new Error('씨앗을 못 찾음 ' + want)
}

  check(/직업 스킬이 없습니다/.test(hA), '암살자는 「직업 스킬이 없습니다」 문구')
  check((hA.match(/class="psl"/g) || []).length === 2, '암살자 패시브는 두 줄(치명타 강화 · 추가 공격)')
  const hP = await battleAs('사제')
  check(/im-ps-ward/.test(hP), '사제 카드에 이탈 방지 배지')
  check(/class="pbn">1</.test(hP), '이탈 방지 남은 횟수 1')
  await L('local st = B.load("key"); B.badd(st.u[1], "atk", 0.15, 3); B.save("key", st); return true')
  const h2 = await display()
  check(/공격 \+15% \(3턴\)/.test(h2), '버프 칩에 남은 턴')
}
// ── 대상 고르는 중인 스킬: 저장을 불러올 때 id 로 번호를 다시 잡는다(옛 저장은 취소)
{
  await battleAs('연금술사')
  const r = await L(`
    local st = B.load("key")
    local i = st.cur
    local slot = B.sk_slot(i, "df") - 1   -- 방어 바로 앞 칸(개인 스킬 자리)
    local id = B.SK[i][slot].id
    local function reload(pend, pid)
      st.pend, st.pid = pend, pid
      B.save("key", st)
      local s2 = B.load("key")
      return { pend = s2.pend, pid = s2.pid }
    end
    local old = reload(slot, nil)                      -- 옛 저장: id 없음
    local same = reload(slot, id)                      -- id 있음 → 같은 스킬
    local moved = reload(1, id)                        -- 번호가 어긋난 저장 → id 로 다시 찾는다
    local gone = reload(slot, "no_such_skill")         -- 목록에 없는 id → 취소
    return { old = old, same = same, moved = moved, gone = gone, slot = slot, id = id }`)
  check(r.old.pend === 0 && !r.old.pid, '옛 저장(st.pid 없음) → 대기 취소')
  check(r.same.pend === r.slot && r.same.pid === r.id, `st.pid 가 있으면 같은 스킬(${r.id})을 가리킨다`)
  check(r.moved.pend === r.slot, '번호가 어긋난 저장도 id 로 다시 찾는다')
  check(r.gone.pend === 0 && !r.gone.pid, '목록에 없는 id → 대기 취소')
  // 대상 확정 때 MP 가 모자라면 취소하고 스킬을 쓰지 않는다
  const mp = await L(`
    local st = B.load("key")
    local i = st.cur
    local slot
    for k, s in ipairs(B.SK[i]) do if s.row == "cls" and (s.mp or 0) > 0 and s.tgt ~= "none" then slot = k; break end end
    if not slot then return { skip = true } end
    st.pend, st.pid = slot, B.SK[i][slot].id
    st.u[i].mp = 0
    local n0 = st.n
    local s2 = B.act(st, "tg_" .. (B.SK[i][slot].tgt == "ally" and i or 5))
    return { pend = s2.pend, pid = s2.pid, moved = s2.n ~= n0, mp = s2.u[i].mp }`)
  check(mp.skip || (mp.pend === 0 && !mp.pid && !mp.moved && mp.mp === 0), 'MP 부족 상태로 대상을 누르면 취소되고 행동이 소비되지 않는다')
}

console.log(`\n${fails === 0 ? '전부 통과' : '실패 ' + fails + '건'} (통과 ${passes})`)
sim.close()
process.exit(fails ? 1 : 0)
