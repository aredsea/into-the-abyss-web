// 적 100종 · 조합기 · 레이드 전투 · 대던전 층 흐름 하네스 — rental-ningen 의 RisuSim(wasmoon)을 읽기 전용으로 빌려 쓴다.
//   node harness_raid.mjs        (먼저 python build.py — dist/bot.lua 를 읽는다)
// 확인: 데이터(적 100 · 계열 30 · 구역 · 균열 · 행동 종류 · 속성) · 모든 적이 어떤 조합에서든 나온다 · 조합 규칙(구역·종류마다 씨앗 200) ·
//       단계 전환 · 분노 · 장기전 · 레이드 감경 · 충전 끊김 · 소환 한도 · 보호 대신 맞기 · 같은 씨앗 = 같은 전투 · 저장 왕복 · 옛 저장 이전 ·
//       화면(nil · NaN · undefined 없음 · 버튼 값 형식 · 메달 · 레이드 패널) · 층 진행(고르기 · 지름길 · 두 단계 · 정산 · 기록) · 균열 의뢰 끝에 핵 ·
//       층이 깊어질수록 줄지 않는 배수(★ 표 · tune.star · 심층)
// ⚠️ DOMPurify·CSS 는 안 거친다 — 그건 Playwright 사진(shots_raid.py)과 실기의 몫이다.
import fs from 'node:fs'
import { fileURLToPath } from 'node:url'
import { mkSim } from './lua_sim.mjs'

const DATA = JSON.parse(fs.readFileSync(new URL('../design/enemies.json', import.meta.url), 'utf8'))
let fails = 0, passes = 0
const check = (ok, msg) => { console.log(`${ok ? 'OK  ' : 'FAIL'} ${msg}`); ok ? passes++ : fails++ }
const { sim, L } = await mkSim()
const SEEDS = +(process.env.SEEDS || 200)

// ---------------------------------------------------------------- 1. 데이터
console.log('=== 1. 데이터 ===')
const E = DATA.enemies
const byId = Object.fromEntries(E.map(e => [e.id, e]))
check(E.length === 100, `적 ${E.length}종 (100)`)
check(new Set(E.map(e => e.id)).size === E.length && E.every(e => /^[a-z][a-z0-9_]*$/.test(e.id)), '적 id 는 ascii snake · 중복 없음')
check(new Set(E.map(e => e.family)).size === 30 && DATA.families.length === 30, '계열 30 전부 쓰인다')
check(DATA.zones.filter(z => z.kind === 'dungeon').length === 9 && DATA.rifts.length === 6, '대던전 구역 9 · 균열 6')
const KINDS = new Set(DATA.kinds), ELEMS = new Set(DATA.elements)
let badAct = 0, badEl = 0
for (const e of E) {
  for (const a of [...e.acts, ...(e.phase2 ? e.phase2.acts : []), ...(e.phase2 && e.phase2.entry ? [e.phase2.entry] : [])]) if (!KINDS.has(a.kind) || !a.id || !a.name) badAct++
  for (const w of [...e.weak, ...((e.phase2 || {}).weak || [])]) if (!ELEMS.has(w)) badEl++
}
check(badAct === 0 && badEl === 0, `행동 종류 · 이름 · id 와 속성이 전부 올바르다 (잘못 ${badAct}/${badEl})`)
check(E.filter(e => e.grade === 'boss').length === 9 && E.filter(e => e.grade === 'core').length === 6 && E.filter(e => e.grade === 'elite').length === 23, '보스 9 · 핵 6 · 정예 23')
const lua = await L(`local n, f = 0, {} for _, id in ipairs(B.FOE_ORDER) do n = n + 1; f[B.FOES[id].fam] = true end local nf = 0 for _ in pairs(f) do nf = nf + 1 end return { n = n, nf = nf, fams = #B.FAMS }`)
check(lua.n === 100 && lua.nf === 30 && lua.fams === 30, `Lua B.FOES ${lua.n}종 · 계열 ${lua.nf} (빌드가 JSON 을 그대로 옮겼다)`)
// 빌드 검증: 모르는 행동 종류 · 속성은 빌드를 실패시킨다(build.py gen_foes) — 별도 프로세스로 확인
{
  const { spawnSync } = await import('node:child_process')
  const bad = JSON.parse(JSON.stringify(DATA))
  bad.enemies[3].acts[0].kind = 'nonsense'
  const tmp = new URL('./_bad_enemies.json', import.meta.url)
  fs.writeFileSync(tmp, JSON.stringify(bad))
  const py = `import sys; sys.path.insert(0, "."); import build; build.ENEMIES_JSON = "_bad_enemies.json"
try:
    build.gen_foes(); print("NOFAIL")
except ValueError as e:
    print("FAIL", e)`
  const HERE = fileURLToPath(new URL('./', import.meta.url))
  const r = spawnSync('python', ['-c', py], { cwd: HERE, encoding: 'utf8', env: { ...process.env, PYTHONIOENCODING: 'utf-8' } })
  const bad2 = JSON.parse(JSON.stringify(DATA)); bad2.enemies[3].weak = ['plasma']
  fs.writeFileSync(tmp, JSON.stringify(bad2))
  const r2 = spawnSync('python', ['-c', py], { cwd: HERE, encoding: 'utf8', env: { ...process.env, PYTHONIOENCODING: 'utf-8' } })
  fs.rmSync(tmp)
  check(/^FAIL/.test(r.stdout.trim()) && /^FAIL/.test(r2.stdout.trim()), '모르는 행동 종류 · 속성은 빌드를 실패시킨다')
}

// ---------------------------------------------------------------- 2. 조합기
console.log('=== 2. 조합 규칙 (구역 · 종류마다 씨앗 ' + SEEDS + ') ===')
const comp = await L(`
  local rep = { bad = {}, reach = {}, n = 0, raidbad = {} }
  local function note(tag, why) rep.bad[tag] = rep.bad[tag] or {}; rep.bad[tag][why] = (rep.bad[tag][why] or 0) + 1 end
  local function mark(fl) for _, id in ipairs(fl) do rep.reach[id] = true; local f = B.FOES[id]; for _, a in ipairs(f.acts) do if a.kind == "summon" then rep.reach[a.enemy] = true end end
    if f.ph2 then for _, a in ipairs(f.ph2.acts) do if a.kind == "summon" then rep.reach[a.enemy] = true end end if f.ph2.entry then rep.reach[f.ph2.entry.enemy] = true end end end end
  local function run(ctxs)
    for _, c in ipairs(ctxs) do
      for s = 1, ${SEEDS} do
        local ctx = {}
        for k, v in pairs(c) do ctx[k] = v end
        ctx.party_n = 1 + (s % 4)
        local r = B.compose(s * 7919 + 13, ctx)
        rep.n = rep.n + 1
        mark(r.fl)
        local tag = (c.zone or c.rift_id) .. "/" .. c.kind
        if c.kind == "battle" or c.kind == "elite" then
          local bad = B.check_comp(r.fl, r.fg, { union = true, umin = r.relaxed and 3 or 4, limit = B.threat_limit(r.order, c.kind == "elite") })
          for _, w in ipairs(bad) do note(tag, w) end
          if r.relaxed then rep.relaxed = (rep.relaxed or 0) + 1; if #r.fl > 2 then note(tag, "완화는 적 둘일 때만") end end
          if c.kind == "elite" then
            local ne = 0
            for k, id in ipairs(r.fl) do if B.FOES[id].grade == "elite" then ne = ne + 1 end end
            if ne ~= 1 then note(tag, "정예가 " .. ne) end
          end
          if #r.fl < 2 or #r.fl > 5 then note(tag, "적 수 " .. #r.fl) end
        else
          local lead = B.FOES[r.fl[1]]
          local want = ({ chief = "elite", boss = "boss", rift_core = "core" })[c.kind]
          if lead.grade ~= want then note(tag, "대표가 " .. lead.grade) end
          if #r.fl > 3 then note(tag, "호위 " .. #r.fl) end
          if c.kind == "boss" and #r.fl ~= 1 then note(tag, "보스 혼자가 아니다") end
          if not r.raid then note(tag, "raid 표식 없음") end
        end
        -- 결정론: 같은 씨앗·같은 ctx = 같은 편성
        if s <= 5 then
          local r2 = B.compose(s * 7919 + 13, ctx)
          if table.concat(r.fl, ",") ~= table.concat(r2.fl, ",") then note(tag, "결정론") end
        end
      end
    end
  end
  local list = {}
  for _, z in ipairs(B.ZONE_ORDER) do
    for _, k in ipairs({ "battle", "elite", "chief", "boss" }) do for o = 1, 3 do list[#list + 1] = { zone = z, kind = k, order = o } end end
  end
  run(list)
  list = {}
  for _, rid in ipairs(B.RIFT_ORDER) do
    list[#list + 1] = { rift_id = rid, kind = "battle", stars = 1 }
    list[#list + 1] = { rift_id = rid, kind = "battle", stars = 2 }
    list[#list + 1] = { rift_id = rid, kind = "elite", stars = 2 }
    list[#list + 1] = { rift_id = rid, kind = "elite", stars = 3 }
    list[#list + 1] = { rift_id = rid, kind = "rift_core", stars = 3 }
  end
  run(list)
  local reach = 0
  local miss = {}
  for _, id in ipairs(B.FOE_ORDER) do if rep.reach[id] then reach = reach + 1 else miss[#miss + 1] = id end end
  return { n = rep.n, bad = rep.bad, reach = reach, miss = miss, relaxed = rep.relaxed or 0 }`)
const badTags = Object.entries(comp.bad || {})
check(badTags.length === 0, `조합 ${comp.n}개: 전열 ≥ 1 · 지원 ≤ 1 · 소환 ≤ 1 · 같은 적 ≤ 3(무리 제외) · 충전 ≤ 1 · 공통 ≤ 1 · 약점 합집합 ≥ 4 · 위협 한도 · 적 수 2~5 · 정예 1 · 보스 구성 · 결정론` +
  (badTags.length ? ' — ' + badTags.slice(0, 6).map(([t, w]) => t + ':' + JSON.stringify(w)).join(' | ') : ''))
console.log(`  (약점 합집합 4가지를 못 지키는 적 둘짜리 편성 ${comp.relaxed}개는 3가지로 완화 — 소규모 파티 · 좁은 풀)`)
check(comp.reach === 100, `모든 적 100종이 어떤 조합(소환 포함)에서든 나온다 (${comp.reach}/100${comp.miss && comp.miss.length ? ' · 빠진 것 ' + comp.miss.join(',') : ''})`)

// ---------------------------------------------------------------- 3. 엔진 규칙
console.log('=== 3. 레이드 규칙 · 행동 ===')
const PRE = `
local function party(L) return ZZ.party("A", L or 20) end
local function mkraid(floor, stage, L, seed)
  local plan = B.floor_plan(floor, L, 4, stage, seed or 7)
  local st = B.fresh(seed or 7, nil, { pm = party(L), el = plan.el, enc = plan.enc })
  B.begin_ev(st)
  return st
end
local function nof(st, i) return B.DEF[i].name end
`
const R = async (code) => L(PRE + code)

// 단계 전환
const ph = await R(`
  local st = mkraid(3, 2, 8, 5)
  local u = st.u[5]
  local before = { wk = table.concat(B.DEF[5].wk, ","), acts = #B.DEF[5].acts, name = B.DEF[5].acts[1].name }
  u.hp = math.floor(u.mhp * 0.49)
  B.after_action(st)
  local alive = 0
  for j = 6, #st.u do if st.u[j].al == 1 and st.u[j].add == 1 then alive = alive + 1 end end
  return { ph = u.ph, wk = table.concat(B.DEF[5].wk, ","), before = before, name2 = B.DEF[5].acts[1].name, ia = u.ia, g = u.g, alive = alive, adds = st.fl[2] .. "," .. st.fl[3], cq = u.cq }`)
check(ph.ph === 2 && ph.wk !== ph.before.wk && ph.name2 === '기말 시험' && ph.ia === 1, `단계 전환(HP 50%): 약점 ${ph.before.wk} → ${ph.wk} · 행동 목록 교체(${ph.name2}) · 행동 순서 처음으로`)
check(ph.alive === 2 && ph.adds === 'apprentice_doll,apprentice_doll' && ph.g === 10000, `단계 전환: 들어서며 견습 인형 둘 소환(${ph.alive}) · 보스 행동 게이지 초기화(${ph.g})`)
// 한 번만 전환
const ph2 = await R(`
  local st = mkraid(3, 2, 8, 5)
  local u = st.u[5]
  u.hp = math.floor(u.mhp * 0.49); B.after_action(st)
  local nev = #st.ev.list
  u.hp = math.floor(u.mhp * 0.2); B.after_action(st)
  return { same = (#st.ev.list == nev), ph = u.ph }`)
check(ph2.same && ph2.ph === 2, '단계 전환은 한 번만 일어난다')
// 우두머리 분노 · 장기전
const rg = await R(`
  local st = mkraid(1, 2, 3, 5)
  local u = st.u[5]
  local m0 = B.foe_atk_mul(st, 5)
  u.hp = math.floor(u.mhp * 0.29)
  local m1 = B.foe_atk_mul(st, 5)
  u.hp = u.mhp
  st.rd = 20; local e20 = B.foe_atk_mul(st, 5)
  st.rd = 25; local e25 = B.foe_atk_mul(st, 5)
  st.rd = 40; local e40 = B.foe_atk_mul(st, 5)
  return { m0 = m0, m1 = m1, e20 = e20, e25 = e25, e40 = e40, grade = B.DEF[5].grade }`)
check(rg.grade === 'chief' && rg.m0 === 1 && Math.abs(rg.m1 - 1.2) < 1e-9, `우두머리 분노: HP 30% 아래 공격 ×${rg.m1} (1.2)`)
check(rg.e20 === 1 && Math.abs(rg.e25 - 1.5) < 1e-9 && Math.abs(rg.e40 - 2) < 1e-9, `장기전: 20라운드까지 ×${rg.e20} · 25라운드 ×${rg.e25} · 40라운드 ×${rg.e40} (+10%/라운드, 최대 +100%)`)
// 레이드 감경: 구역 보스만 효과를 절반으로
const rd = await R(`
  local st = mkraid(3, 2, 8, 5)
  local bu = st.u[5]
  B.badd(bu, "adn", 0.2, 2)
  local a = bu.bf.adn.v
  local g0 = bu.g; bu.g = 0; local dly = 10000 * B.rdm(bu)
  local st2 = mkraid(1, 2, 3, 5)
  local cu = st2.u[5]
  B.badd(cu, "adn", 0.2, 2)
  local ev = { { b = bu.rd, bossv = a }, { c = cu.rd, chiefv = cu.bf.adn.v, rdm = B.rdm(cu) } }
  return { bossrd = bu.rd, bossv = a, dly = dly, chiefrd = cu.rd, chiefv = cu.bf.adn.v }`)
check(rd.bossrd === 1 && Math.abs(rd.bossv - 0.1) < 1e-9 && rd.dly === 5000, `레이드 감경(구역 보스): 약화 20% → ${rd.bossv * 100}% · 지연 10000 → ${rd.dly}`)
check(rd.chiefrd === 0 && Math.abs(rd.chiefv - 0.2) < 1e-9, `우두머리는 감경하지 않는다 (약화 ${rd.chiefv * 100}%)`)
// 충전 → 심연 고갈로 끊기 · 큰 기술
const cc = await R(`
  local st = mkraid(1, 2, 8, 11)    -- 마도 집행 인형 또는 수석 조교
  local e = nil
  for seed = 1, 60 do
    local s2 = mkraid(1, 2, 8, seed)
    if B.FOES[s2.fl[1]].id == "magic_enforcer" then st = s2; e = true; break end
  end
  local u, d = st.u[5], B.DEF[5]
  -- 다음 행동이 「과열(충전)」이 되게 하고, 적 차례를 직접 돌려 충전 → 큰 기술
  u.ia = 2
  B.begin_ev(st)
  B.rbegin(st)
  local hp0 = {}
  for i = 1, 4 do hp0[i] = st.u[i].hp end
  B.enemy_turn(st, 5)           -- 과열(충전)
  local cq1, ia1 = u.cq, u.ia
  u.ia = 3                      -- (enemy_turn 이 ia 를 3 으로 올렸다)
  local dmg0 = 0
  for i = 1, 4 do dmg0 = dmg0 + (hp0[i] - st.u[i].hp) end
  -- 충전 중 심연을 0 으로 깎는다
  B.setf(st, 5, "ay", 1, st.ev.T)
  B.abyss(st, 1, 5, 99, "none", st.ev.T)
  local dr = u.dr
  B.lost_turn(st, 5)
  local dmg1 = 0
  for i = 1, 4 do dmg1 = dmg1 + (hp0[i] - st.u[i].hp) end
  return { found = e, cq1 = cq1, name = B.DEF[5].acts[ia1].name, dr = dr, cq2 = u.cq, ia2 = u.ia, dmg0 = dmg0, dmg1 = dmg1, hp = u.hp, mhp = u.mhp, cut = (function() for _, x in ipairs(st.log) do if x.h:find("끊겼다", 1, true) then return true end end return false end)() }`)
check(cc.found && cc.cq1 === 1 && cc.name === '증기 분출', `충전: 과열 뒤 다음 행동이 큰 기술(${cc.name}) · 충전 표식 ${cc.cq1}`)
check(cc.dr === 1 && cc.cq2 === 0 && cc.ia2 === 4 && cc.dmg1 === cc.dmg0 && cc.cut, `충전 끊김: 심연 0 → 고갈 · 큰 기술 취소(행동 ${cc.ia2}번으로) · 아군 피해 증가 없음 · 과열 풀려 스스로 피해(HP ${cc.hp}/${cc.mhp})`)
const bigno = await R(`
  local st
  for seed = 1, 60 do
    local s2 = mkraid(1, 2, 8, seed)
    if B.FOES[s2.fl[1]].id == "magic_enforcer" then st = s2; break end
  end
  local u = st.u[5]
  u.ia = 2; B.begin_ev(st); B.rbegin(st)
  local hp0 = 0
  for i = 1, 4 do hp0 = hp0 + st.u[i].hp end
  B.enemy_turn(st, 5)   -- 충전
  B.enemy_turn(st, 5)   -- 큰 기술(전체)
  local hp1 = 0
  local hit = 0
  for i = 1, 4 do hp1 = hp1 + st.u[i].hp; if st.u[i].hp < st.u[i].mhp then hit = hit + 1 end end
  return { dmg = hp0 - hp1, hit = hit, cq = u.cq }`)
check(bigno.dmg > 0 && bigno.hit >= 3 && bigno.cq === 0, `충전을 끊지 않으면 큰 기술이 전체에 들어간다 (피해 ${bigno.dmg}, ${bigno.hit}명)`)
// 소환 한도
const sm = await R(`
  local st = mkraid(3, 2, 8, 5)
  local made = 0
  for k = 1, 6 do if B.summon_one(st, 5, "apprentice_doll") then made = made + 1 end end
  local adds = 0
  for j = 6, #st.u do if st.u[j].al == 1 and st.u[j].add == 1 then adds = adds + 1 end end
  -- 하나가 쓰러지면 다시 소환할 수 있다(칸 재사용)
  st.u[6].al = 0; st.u[6].hp = 0
  local again = B.summon_one(st, 5, "gate_statue")
  local hpfrac = st.u[7].mhp / st.u[5].mhp
  return { made = made, adds = adds, again = again and true or false, role = B.DEF[7].srole, grade = B.DEF[7].grade, hpfrac = hpfrac }`)
check(sm.made === 2 && sm.adds === 2 && sm.again, `소환 한도: 하수인 최대 2 (여섯 번 시도 → ${sm.made}) · 하나 쓰러지면 다시 가능`)
check(sm.grade === 'minion' && sm.role === '하수인' && sm.hpfrac < 0.15, `하수인: 등급 minion · 역할 배수 하수인 · 보스 HP 의 ${(sm.hpfrac * 100).toFixed(1)}%`)
// 보호 대신 맞기
const pr = await R(`
  local st
  for seed = 1, 80 do
    local s2 = mkraid(5, 1, 12, seed)
    local has = false
    for _, id in ipairs(s2.fl) do if B.FOES[id].acts[1].kind == "protect" then has = true end end
    if has and #s2.fl >= 3 then st = s2; break end
  end
  if not st then return { found = false } end
  local p, t
  for j = 5, #st.u do if st.u[j].empty ~= 1 and B.DEF[j].acts[1].kind == "protect" then p = j end end
  for j = 5, #st.u do if st.u[j].empty ~= 1 and j ~= p then t = j; break end end
  st.u[p].pt = t
  st.u[p].sh = 0; st.u[t].sh = 0   -- 보호막 없이 시작(첫 차례 순서가 바뀌면 보호하는 적이 먼저 자기 보호막을 칠 수 있다 — 아군 속도 보너스 수치에 흔들리지 않게)
  local php, thp = st.u[p].hp, st.u[t].hp
  B.begin_ev(st)
  local res = B.attack(st, 1, t, { m = 1.0, brk = 0, name = "일반 공격", vb = "베었다", hit = 0.4 })
  return { found = true, redirected = (res.i == p), pdmg = php - st.u[p].hp, tdmg = thp - st.u[t].hp }`)
check(pr.found && pr.redirected && pr.pdmg > 0 && pr.tdmg === 0, `보호: 보호 중인 적이 대신 맞는다 (보호자 피해 ${pr.pdmg} · 표적 피해 ${pr.tdmg})`)
// 같은 씨앗 = 같은 전투 · 저장 왕복
const det = await R(`
  local function fight() return ZZ.floor_fight({ mix = "A", L = 9, floor = 6, stage = 2, seed = 4242, mode = "smart" }) end

    local dpf = B.FOES[bid].deep

      local plan = B.floor_plan(f, 40, 1, 2, 11)
      local st = B.fresh(1, nil, { pm = ZZ.party("A", 40), el = 40, enc = plan.enc })
      local d = B.DEF[5]
      local s = B.foe_stats(st, d)
      local mh = s.hp / (B.ENEMY_STD.hp * B.fc("atk", 40) * B.ROLES[d.srole].hp * B.GRADES[d.grade].hp)
      local fh = B.floor_final(f, "boss")
      local want = fh * ((f > 27) and ((dpf and dpf.hp) or 1) or (B.ZONES[z].tune.boss_hp or 1))   -- 1~27층은 구역 보정 boss_hp(10-10 흐름 재튜닝) · 심층은 보스별 계수
      local ea = B.EX.EARLY
      if ea and f < ea.to then want = want * ea.hp ^ (1 - ((f - 1) / (ea.to - 1)) ^ (ea.curve or 1)) end   -- 입문 구간(7층 전) 보정
      out.n = out.n + 1
      if d.foe ~= bid then out.bad[#out.bad + 1] = bid .. " " .. f .. "층 보스가 아님 " .. tostring(d.foe) end
      if math.abs(mh - want) > 1e-6 * want then out.bad[#out.bad + 1] = bid .. " " .. f .. "층 HP 배율 " .. mh .. " ≠ " .. want end
      if c >= 1 then
        if mh < last - 1e-9 then out.bad[#out.bad + 1] = bid .. " " .. f .. "층 순환 간 HP 배율이 줄었다" end
        last = mh
      end
    end
  end
  for id, f in pairs(B.FOES) do
    if f.deep and f.grade ~= "boss" then out.bad[#out.bad + 1] = id .. " deep 은 구역 보스만" end
